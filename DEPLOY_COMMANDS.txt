# Fly.io backend deployment commands
# Fill in placeholders locally. Never paste real secrets into this file.

# 1. Authenticate
fly auth login

# 2. Create or select the app
cd backend
fly launch --no-deploy

# 3. Create persistent upload storage.
# Region should match backend/fly.toml primary_region.
fly volumes create usflix_uploads --size 10 --region sjc

# 4. Set required secrets
fly secrets set DATABASE_URL="postgresql://user:password@host:5432/dbname?sslmode=require"
fly secrets set JWT_SECRET="replace-with-output-from-node-scripts-generate-jwt-secret"
fly secrets set ADMIN_USERNAME="admin"
fly secrets set ADMIN_PASSWORD="replace-with-a-strong-admin-password"
fly secrets set FRONTEND_URL="https://your-frontend.example.com"

# 5. Optional production secrets
fly secrets set OPENWEATHER_API_KEY="replace-if-weather-is-enabled"
fly secrets set COOKIE_SAME_SITE="none"

# 6. Deploy and verify
fly deploy
fly status
curl https://your-fly-app.fly.dev/api/health

# Useful operations
fly logs
fly secrets list
fly apps restart
fly ssh console
