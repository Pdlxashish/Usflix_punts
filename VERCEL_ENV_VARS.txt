# Frontend environment variables
#
# This project is configured primarily for Cloudflare Workers. If you deploy a
# frontend build to another host such as Vercel, do not add backend-only secrets
# there. The browser-facing frontend should only need VITE_* variables.

# Required only for split frontend/API domains:
VITE_API_URL=https://your-backend.example.com

# Leave VITE_API_URL empty for same-origin deployments where /api and /uploads
# are proxied to the backend by nginx, Cloudflare, or another edge proxy.
